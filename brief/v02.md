## 📋 **Table Columns (Node Inspection):**

```typescript
interface NodeTransaction {
  date: Date;           // When it happened
  changes: number;      // +200k or -50k
  value: number;        // Current balance after this change
  diff?: number;        // Optional: previous - current (for visualization)
  input_node?: string;  // Where it came FROM
  output_node?: string; // Where it went TO
  node_assigned: string; // Which node owns this transaction
}

// EXAMPLE TABLE VIEW:
┌────────┬──────────┬─────────┬──────────┬─────────────┬──────────────┬────────────────┐
```

## 🎯 **Implementation Structure:**

```typescript
// MODE TOGGLE
const [graphMode, setGraphMode] = useState<'simple' | 'irl'>('simple');

// SELECTED NODE
const [selectedNode, setSelectedNode] = useState<Node | null>(null);
const [isMobile, setIsMobile] = useState(window.innerWidth < 768);

// COMPONENTS
export function DualModeGraphView() {
  return (
    <div className="flex gap-4">
      {/* LEFT: Graph */}
      <div className="flex-1">
        <ModeToggle graphMode={graphMode} setGraphMode={setGraphMode} />
        
        {graphMode === 'simple' ? (
          <SimpleLinearGraph onNodeSelect={setSelectedNode} />
        ) : (
          <IRLComplexGraph onNodeSelect={setSelectedNode} />
        )}
      </div>

      {/* RIGHT: Sidebar or Popup */}
      {selectedNode && (
        isMobile ? (
          <NodeInspectPopup node={selectedNode} />
        ) : (
          <NodeInspectSidebar node={selectedNode} />
        )
      )}
    </div>
  );
}

// SIDEBAR (Desktop/Tablet)
export function NodeInspectSidebar({ node }: { node: Node }) {
  return (
    <div className="w-96 bg-white border-l p-6 overflow-y-auto">
      <h2 className="text-2xl font-bold">{node.label}</h2>
      <p className="text-lg text-gray-600">Balance: Rp {node.balance?.toLocaleString()}</p>
      
      {/* TRANSACTION TABLE */}
      <div className="mt-6">
        <h3 className="font-semibold mb-4">Transaction History</h3>
        <table className="w-full text-sm">
          <thead>
            <tr className="border-b">
              <th className="text-left">Date</th>
              <th className="text-right">Changes</th>
              <th className="text-right">Value</th>
              <th className="text-right">Diff</th>
              <th className="text-left">Input</th>
              <th className="text-left">Output</th>
            </tr>
          </thead>
          <tbody>
            {node.transactions.map((tx, i) => (
              <tr key={i} className="border-b hover:bg-gray-50">
                <td>{tx.date.toLocaleDateString()}</td>
                <td className={`text-right font-semibold ${tx.changes > 0 ? 'text-green-600' : 'text-red-600'}`}>
                  {tx.changes > 0 ? '+' : ''}{(tx.changes / 1000).toFixed(0)}k
                </td>
                <td className="text-right">{(tx.value / 1000).toFixed(0)}k</td>
                <td className="text-right text-gray-500">{tx.diff}</td>
                <td className="text-xs text-gray-600">{tx.input_node}</td>
                <td className="text-xs text-gray-600">{tx.output_node}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      {/* INPUT/OUTPUT NODES */}
      <div className="mt-6 grid grid-cols-2 gap-4">
        <div>
          <h4 className="font-semibold mb-2">Inputs</h4>
          <ul className="text-sm space-y-1">
            {node.inputs?.map((input) => (
              <li key={input.id} className="text-blue-600 cursor-pointer hover:underline">
                {input.label}
              </li>
            ))}
          </ul>
        </div>
        <div>
          <h4 className="font-semibold mb-2">Outputs</h4>
          <ul className="text-sm space-y-1">
            {node.outputs?.map((output) => (
              <li key={output.id} className="text-orange-600 cursor-pointer hover:underline">
                {output.label}
              </li>
            ))}
          </ul>
        </div>
      </div>
    </div>
  );
}

// POPUP (Mobile)
export function NodeInspectPopup({ node }: { node: Node }) {
  const [isOpen, setIsOpen] = useState(true);

  return (
    <Dialog open={isOpen} onOpenChange={setIsOpen}>
      <DialogContent className="max-w-lg">
        <DialogHeader>
          <DialogTitle>{node.label}</DialogTitle>
          <DialogDescription>
            Balance: Rp {node.balance?.toLocaleString()}
          </DialogDescription>
        </DialogHeader>

        {/* TABLE IN POPUP */}
        <div className="overflow-y-auto max-h-96">
          <table className="w-full text-xs">
            <thead>
              <tr className="border-b sticky top-0 bg-gray-50">
                <th className="text-left p-2">Date</th>
                <th className="text-right p-2">Changes</th>
                <th className="text-right p-2">Value</th>
                <th className="text-left p-2">Node</th>
              </tr>
            </thead>
            <tbody>
              {node.transactions.map((tx, i) => (
                <tr key={i} className="border-b">
                  <td className="p-2">{tx.date.toLocaleDateString()}</td>
                  <td className={`text-right p-2 font-semibold ${tx.changes > 0 ? 'text-green-600' : 'text-red-600'}`}>
                    {tx.changes > 0 ? '+' : ''}{(tx.changes / 1000).toFixed(0)}k
                  </td>
                  <td className="text-right p-2">{(tx.value / 1000).toFixed(0)}k</td>
                  <td className="text-xs p-2 text-gray-600">{tx.node_assigned}</td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </DialogContent>
    </Dialog>
  );
}
```

```typescript
  window.addEventListener('resize', handleResize);
  return () => window.removeEventListener('resize', handleResize);
}, []);

// Desktop/Tablet (≥768px): Sidebar fixed on right
```

```typescript
    .doc(selectedNode.id)
    .onSnapshot((doc) => {
      setSelectedNode(doc.data());
      // Table auto-updates
    });

  return () => unsubscribe();
```
